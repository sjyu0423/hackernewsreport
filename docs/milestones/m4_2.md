```bash
import json
print(json.dumps(inspect_scheduler(), indent=2, default=str))
PY

# Re-check agents (do not unload/replace)
launchctl print gui/$(id -u)/com.qwennews.hourly | rg 'runs|state|last exit'
wc -c data/logs/hourly.*.log data/logs/daily.*.log
```
